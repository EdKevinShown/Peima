## 4. Step 1 — 找 clean viewer candidates

**环境：** dev/staging · `docker exec -i peima-postgres psql -U user -d peima`  
**保存：** `tmp/p76-r7j-viewer-candidates.sql`

**目的：** 列出**自身画像完整**、可作为 Route C **viewer** 的 user（**不要求**已有 pool；用于 C-A/C-B 选型）。

```sql
-- P7.6-r7j Step 1: clean viewer candidates (READ-ONLY)

WITH profile_dims AS (
  SELECT up."userId",
    (
      (CASE WHEN up."relationshipPace"    IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."emotionalStability" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."communicationStyle" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."conflictHandling"   IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."lifePace"           IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."socialNeed"         IS NOT NULL THEN 1 ELSE 0 END)
    ) AS non_null_core_dim_count
  FROM user_profile up
),
viewer_vision AS (
  SELECT ui."userId",
    BOOL_OR(ui."detectionStatus" = 'passed' AND ui."detectionScoreJson" IS NOT NULL) AS has_live_vision,
    BOOL_OR(ui."detectionStatus" = 'passed') AS has_passed_image,
    (
      SELECT ui2."detectionScoreJson"->'vision'->>'sourceVersion'
      FROM user_images ui2
      WHERE ui2."userId" = ui."userId" AND ui2."detectionStatus" = 'passed'
        AND ui2."detectionScoreJson" IS NOT NULL
      ORDER BY ui2."updatedAt" DESC LIMIT 1
    ) AS vision_source_version,
    (
      SELECT jsonb_array_length(COALESCE(ui2."detectionScoreJson"->'vision'->'photoVisualTags', '[]'::jsonb))
      FROM user_images ui2
      WHERE ui2."userId" = ui."userId" AND ui2."detectionStatus" = 'passed'
        AND ui2."detectionScoreJson" IS NOT NULL
      ORDER BY ui2."updatedAt" DESC LIMIT 1
    ) AS style_tags_count
  FROM user_images ui
  GROUP BY ui."userId"
),
pool_counts AS (
  SELECT opp."userId",
    COUNT(DISTINCT opp."id") FILTER (WHERE opp."status" = 'active') AS existing_pool_count,
    COUNT(DISTINCT oppi."candidateUserId") FILTER (WHERE opp."status" = 'active') AS active_candidate_count
  FROM onboarding_photo_preview_pools opp
  LEFT JOIN onboarding_photo_preview_pool_items oppi ON oppi."poolId" = opp."id"
  GROUP BY opp."userId"
),
latest_match AS (
  SELECT DISTINCT ON (mr."userId") mr."userId",
    (mr."matchInsights" ? 'rrmV2Top2Selector') AS has_m6_rrm_top2
  FROM match_results mr
  ORDER BY mr."userId", mr."updatedAt" DESC, mr."createdAt" DESC
),
seed_flags AS (
  SELECT u."id" AS "userId",
    (
      EXISTS (SELECT 1 FROM user_profile up WHERE up."userId" = u."id" AND up."id" LIKE 'dev-p76-%')
      OR EXISTS (SELECT 1 FROM user_preferences pref WHERE pref."userId" = u."id" AND pref."id" LIKE 'dev-p76-%')
    ) AS seed_dependent
  FROM users u
)
SELECT
  u."id" AS "userId",
  COALESCE(vv.has_live_vision, false) AS "hasLiveVision",
  COALESCE(pd.non_null_core_dim_count, 0) AS "nonNull20DCount",
  (pref."userId" IS NOT NULL) AS "hasPreference",
  COALESCE(vv.has_passed_image, false) AS "hasPassedImage",
  vv.vision_source_version AS "visionSourceVersion",
  COALESCE(vv.style_tags_count, 0) AS "styleTags",
  u."gender",
  pref."minAge",
  pref."maxAge",
  pref."preferredCities",
  COALESCE(pc.existing_pool_count, 0) AS "existingPoolCount",
  COALESCE(pc.active_candidate_count, 0) AS "activeCandidateCount",
  (lm."userId" IS NOT NULL) AS "hasMatchResult",
  COALESCE(lm.has_m6_rrm_top2, false) AS "hasM6RrmTop2",
  COALESCE(sf.seed_dependent, false) AS "seedDependent"
FROM users u
LEFT JOIN viewer_vision vv ON vv."userId" = u."id"
LEFT JOIN profile_dims pd ON pd."userId" = u."id"
LEFT JOIN user_preferences pref ON pref."userId" = u."id"
LEFT JOIN pool_counts pc ON pc."userId" = u."id"
LEFT JOIN latest_match lm ON lm."userId" = u."id"
LEFT JOIN seed_flags sf ON sf."userId" = u."id"
WHERE u."id" NOT IN (
  'cmo7ksq8s00006znosryc9k0n',
  'cmp70ft7a001e6znksjb2jo03',
  'cmfemn002000306z64seed0002',
  'cmfemn003000506z64seed0003'
)
  AND COALESCE(vv.has_live_vision, false) = true
  AND COALESCE(pd.non_null_core_dim_count, 0) >= 4
  AND pref."userId" IS NOT NULL
  AND NOT COALESCE(sf.seed_dependent, false)
ORDER BY "activeCandidateCount" DESC, "hasMatchResult" DESC, "hasM6RrmTop2" DESC, "nonNull20DCount" DESC
LIMIT 50;
```

### 4.1 Step 1 产出

| 产出 | 说明 |
|------|------|
| `staging-cohort-viewer-candidates.csv` | §4 SQL 导出 |
| **优先 viewer** | `activeCandidateCount` 高 · `seedDependent=false` · 非 r7c5 pool |
| **次优先** | `activeCandidateCount=0` 但画像完整 → 进入 §6 建 pool |

## 5. Step 2 — 找 reusable candidate candidates

**保存：** `tmp/p76-r7j-reusable-candidates.sql`  
**目的：** 为 §6 建 pool 或补池提供**合格候选**素材（**不等于** viewer）。

```sql
-- P7.6-r7j Step 2: reusable candidate users (READ-ONLY)

WITH profile_dims AS (
  SELECT up."userId",
    (
      (CASE WHEN up."relationshipPace"    IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."emotionalStability" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."communicationStyle" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."conflictHandling"   IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."lifePace"           IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."socialNeed"         IS NOT NULL THEN 1 ELSE 0 END)
    ) AS non_null_core_dim_count
  FROM user_profile up
),
candidate_vision AS (
  SELECT ui."userId",
    BOOL_OR(ui."detectionStatus" = 'passed' AND ui."detectionScoreJson" IS NOT NULL) AS has_vision,
    (
      SELECT ui2."detectionScoreJson"->'vision'->>'provider'
      FROM user_images ui2
      WHERE ui2."userId" = ui."userId" AND ui2."detectionStatus" = 'passed'
      ORDER BY ui2."updatedAt" DESC LIMIT 1
    ) AS vision_provider,
    (
      SELECT ui2."detectionScoreJson"->'vision'->>'sourceVersion'
      FROM user_images ui2
      WHERE ui2."userId" = ui."userId" AND ui2."detectionStatus" = 'passed'
        AND ui2."detectionScoreJson" IS NOT NULL
      ORDER BY ui2."updatedAt" DESC LIMIT 1
    ) AS source_version,
    (
      SELECT jsonb_array_length(COALESCE(ui2."detectionScoreJson"->'vision'->'photoVisualTags', '[]'::jsonb))
      FROM user_images ui2
      WHERE ui2."userId" = ui."userId" AND ui2."detectionStatus" = 'passed'
        AND ui2."detectionScoreJson" IS NOT NULL
      ORDER BY ui2."updatedAt" DESC LIMIT 1
    ) AS photo_visual_tags_count,
    MAX(ui."reviewStatus") FILTER (WHERE ui."detectionStatus" = 'passed') AS review_status,
    MAX(ui."detectionStatus") AS detection_status
  FROM user_images ui
  GROUP BY ui."userId"
),
seed_flags AS (
  SELECT u."id" AS "userId",
    (
      EXISTS (SELECT 1 FROM user_profile up WHERE up."userId" = u."id" AND up."id" LIKE 'dev-p76-%')
      OR EXISTS (SELECT 1 FROM user_preferences pref WHERE pref."userId" = u."id" AND pref."id" LIKE 'dev-p76-%')
    ) AS seed_dependent
  FROM users u
)
SELECT
  u."id" AS "candidateUserId",
  COALESCE(cv.has_vision, false) AS "hasVision",
  cv.vision_provider AS "visionProvider",
  cv.source_version AS "sourceVersion",
  COALESCE(cv.photo_visual_tags_count, 0) AS "photoVisualTagsCount",
  COALESCE(pd.non_null_core_dim_count, 0) AS "nonNull20DCount",
  (pref."userId" IS NOT NULL) AS "hasPreference",
  cv.review_status AS "reviewStatus",
  cv.detection_status AS "detectionStatus",
  u."gender",
  COALESCE(sf.seed_dependent, false) AS "seedDependent"
FROM users u
LEFT JOIN candidate_vision cv ON cv."userId" = u."id"
LEFT JOIN profile_dims pd ON pd."userId" = u."id"
LEFT JOIN user_preferences pref ON pref."userId" = u."id"
LEFT JOIN seed_flags sf ON sf."userId" = u."id"
WHERE u."id" NOT IN (
  'cmo7ksq8s00006znosryc9k0n',
  'cmp70ft7a001e6znksjb2jo03',
  'cmfemn002000306z64seed0002',
  'cmfemn003000506z64seed0003'
)
  AND COALESCE(cv.has_vision, false) = true
  AND COALESCE(pd.non_null_core_dim_count, 0) >= 4
  AND pref."userId" IS NOT NULL
  AND (cv.review_status IS NULL OR cv.review_status NOT IN ('rejected', 'blocked'))
  AND cv.detection_status = 'passed'
ORDER BY "photoVisualTagsCount" DESC, "nonNull20DCount" DESC
LIMIT 100;
```
