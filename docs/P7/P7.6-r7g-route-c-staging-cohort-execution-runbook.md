## 4. Cohort readiness SQL（只读）

**环境：** dev/staging · `docker exec -i peima-postgres psql -U user -d peima`  
**保存：** `tmp/p76-r7g-cohort-readiness.sql`（执行时）

基于 [r7c §4](./P7.6-r7c-multi-sample-audit-execution-runbook.md)，增加 Route C 标记列：

```sql
-- P7.6-r7g: Route C staging viewer readiness (READ-ONLY)

WITH profile_dims AS (
  SELECT up."userId",
    (
      (CASE WHEN up."relationshipPace"    IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."emotionalStability" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."communicationStyle" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."conflictHandling"   IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."lifePace"           IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."socialNeed"         IS NOT NULL THEN 1 ELSE 0 END)
    ) AS non_null_core_dim_count
  FROM user_profile up
),
viewer_vision AS (
  SELECT ui."userId",
    BOOL_OR(ui."detectionStatus" = 'passed' AND ui."detectionScoreJson" IS NOT NULL) AS has_live_vision
  FROM user_images ui
  GROUP BY ui."userId"
),
onboarding_pool AS (
  SELECT opp."userId" AS viewer_user_id, opp."id" AS pool_id,
    COUNT(DISTINCT oppi."candidateUserId") AS candidate_pool_count
  FROM onboarding_photo_preview_pools opp
  JOIN onboarding_photo_preview_pool_items oppi ON oppi."poolId" = opp."id"
  WHERE opp."status" = 'active'
  GROUP BY opp."userId", opp."id"
),
candidate_vision AS (
  SELECT oppi."userId" AS viewer_user_id,
    COUNT(DISTINCT oppi."candidateUserId") FILTER (
      WHERE EXISTS (
        SELECT 1 FROM user_images ci
        WHERE ci."userId" = oppi."candidateUserId"
          AND ci."detectionStatus" = 'passed' AND ci."detectionScoreJson" IS NOT NULL
      )
    ) AS candidate_with_vision_count
  FROM onboarding_photo_preview_pool_items oppi
  JOIN onboarding_photo_preview_pools opp ON opp."id" = oppi."poolId" AND opp."status" = 'active'
  GROUP BY oppi."userId"
),
candidate_profile AS (
  SELECT oppi."userId" AS viewer_user_id,
    COUNT(DISTINCT oppi."candidateUserId") FILTER (WHERE pd_c.non_null_core_dim_count >= 4) AS candidate_with_20d_profile_count
  FROM onboarding_photo_preview_pool_items oppi
  JOIN onboarding_photo_preview_pools opp ON opp."id" = oppi."poolId" AND opp."status" = 'active'
  LEFT JOIN profile_dims pd_c ON pd_c."userId" = oppi."candidateUserId"
  GROUP BY oppi."userId"
),
candidate_pref AS (
  SELECT oppi."userId" AS viewer_user_id,
    COUNT(DISTINCT oppi."candidateUserId") FILTER (WHERE cp."userId" IS NOT NULL) AS candidate_with_preference_count
  FROM onboarding_photo_preview_pool_items oppi
  JOIN onboarding_photo_preview_pools opp ON opp."id" = oppi."poolId" AND opp."status" = 'active'
  LEFT JOIN user_preferences cp ON cp."userId" = oppi."candidateUserId"
  GROUP BY oppi."userId"
),
latest_match AS (
  SELECT DISTINCT ON (mr."userId") mr."userId",
    (mr."matchInsights" ? 'rrmV2Top2Selector') AS has_m6_rrm_top2
  FROM match_results mr
  ORDER BY mr."userId", mr."updatedAt" DESC, mr."createdAt" DESC
),
seed_flags AS (
  SELECT u."id" AS "userId",
    (
      EXISTS (SELECT 1 FROM user_profile up WHERE up."userId" = u."id" AND up."id" LIKE 'dev-p76-%')
      OR EXISTS (SELECT 1 FROM user_preferences pref WHERE pref."userId" = u."id" AND pref."id" LIKE 'dev-p76-%')
    ) AS seed_dependent
  FROM users u
),
synthetic_pool AS (
  SELECT opp."userId" AS viewer_user_id,
    BOOL_OR(opp."id" LIKE 'dev-p76-r7c5-%' OR opp."id" LIKE 'dev-p76-r7c3-%') AS synthetic_pool
  FROM onboarding_photo_preview_pools opp
  WHERE opp."status" = 'active'
  GROUP BY opp."userId"
)
SELECT
  u."id" AS "viewerUserId",
  COALESCE(vv.has_live_vision, false) AS "hasLiveVision",
  COALESCE(pd.non_null_core_dim_count, 0) AS "nonNull20DCount",
  (pref."userId" IS NOT NULL) AS "hasPreference",
  COALESCE(op.candidate_pool_count, 0) AS "candidatePoolCount",
  COALESCE(cv.candidate_with_vision_count, 0) AS "candidateWithVisionCount",
  COALESCE(cprof.candidate_with_20d_profile_count, 0) AS "candidateWith20DProfileCount",
  COALESCE(cpref.candidate_with_preference_count, 0) AS "candidateWithPreferenceCount",
  (lm."userId" IS NOT NULL) AS "hasMatchResult",
  COALESCE(lm.has_m6_rrm_top2, false) AS "hasM6RrmTop2",
  COALESCE(sf.seed_dependent, false) AS "seedDependent",
  COALESCE(sp.synthetic_pool, false) AS "syntheticPool",
  (
    COALESCE(vv.has_live_vision, false)
    AND COALESCE(pd.non_null_core_dim_count, 0) >= 4
    AND pref."userId" IS NOT NULL
    AND COALESCE(op.candidate_pool_count, 0) >= 6
    AND COALESCE(cv.candidate_with_vision_count, 0) >= 6
    AND COALESCE(cprof.candidate_with_20d_profile_count, 0) >= 6
    AND COALESCE(cpref.candidate_with_preference_count, 0) >= 6
    AND NOT COALESCE(sf.seed_dependent, false)
    AND NOT COALESCE(sp.synthetic_pool, false)
  ) AS "recommendedForRouteC"
FROM users u
LEFT JOIN viewer_vision vv ON vv."userId" = u."id"
LEFT JOIN profile_dims pd ON pd."userId" = u."id"
LEFT JOIN user_preferences pref ON pref."userId" = u."id"
LEFT JOIN onboarding_pool op ON op.viewer_user_id = u."id"
LEFT JOIN candidate_vision cv ON cv.viewer_user_id = u."id"
LEFT JOIN candidate_profile cprof ON cprof.viewer_user_id = u."id"
LEFT JOIN candidate_pref cpref ON cpref.viewer_user_id = u."id"
LEFT JOIN latest_match lm ON lm."userId" = u."id"
LEFT JOIN seed_flags sf ON sf."userId" = u."id"
LEFT JOIN synthetic_pool sp ON sp.viewer_user_id = u."id"
WHERE u."id" NOT IN (
  'cmo7ksq8s00006znosryc9k0n',
  'cmp70ft7a001e6znksjb2jo03',
  'cmfemn002000306z64seed0002',
  'cmfemn003000506z64seed0003'
)
ORDER BY "recommendedForRouteC" DESC, "hasMatchResult" DESC, "hasM6RrmTop2" DESC, "nonNull20DCount" DESC
LIMIT 50;
```

| `hasPreference` | `true` |

## 5. Candidate readiness SQL（只读）

对每个入选 `viewerUserId` 执行（替换 `:viewer_id`）：

```sql
-- P7.6-r7g: per-viewer candidate readiness (READ-ONLY)

WITH profile_dims AS (
  SELECT up."userId",
    (
      (CASE WHEN up."relationshipPace" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."emotionalStability" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."communicationStyle" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."conflictHandling" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."lifePace" IS NOT NULL THEN 1 ELSE 0 END) +
      (CASE WHEN up."socialNeed" IS NOT NULL THEN 1 ELSE 0 END)
    ) AS non_null_core_dim_count
  FROM user_profile up
)
SELECT
  oppi."candidateUserId",
  EXISTS (
    SELECT 1 FROM user_images ui
    WHERE ui."userId" = oppi."candidateUserId"
      AND ui."detectionStatus" = 'passed' AND ui."detectionScoreJson" IS NOT NULL
  ) AS "hasVision",
  (
    SELECT ui."detectionScoreJson"->'vision'->>'provider'
    FROM user_images ui
    WHERE ui."userId" = oppi."candidateUserId" AND ui."detectionStatus" = 'passed'
    ORDER BY ui."updatedAt" DESC LIMIT 1
  ) AS "visionProvider",
  (
    SELECT ui."detectionScoreJson"->'vision'->>'sourceVersion'
    FROM user_images ui
    WHERE ui."userId" = oppi."candidateUserId" AND ui."detectionStatus" = 'passed'
    ORDER BY ui."updatedAt" DESC LIMIT 1
  ) AS "visionSourceVersion",
  (
    SELECT jsonb_array_length(COALESCE(ui."detectionScoreJson"->'vision'->'photoVisualTags', '[]'::jsonb))
    FROM user_images ui
    WHERE ui."userId" = oppi."candidateUserId" AND ui."detectionStatus" = 'passed'
    ORDER BY ui."updatedAt" DESC LIMIT 1
  ) AS "photoVisualTagsCount",
  COALESCE(pd.non_null_core_dim_count, 0) AS "nonNull20DCount",
  (pref."userId" IS NOT NULL) AS "hasPreference",
  CASE
    WHEN oppi."candidateUserId" = ':viewer_id' THEN 'SELF_IN_POOL'
    WHEN NOT EXISTS (
      SELECT 1 FROM user_images ui
      WHERE ui."userId" = oppi."candidateUserId"
        AND ui."detectionStatus" = 'passed' AND ui."detectionScoreJson" IS NOT NULL
    ) THEN 'NO_VISION'
    WHEN COALESCE(pd.non_null_core_dim_count, 0) < 4 THEN 'PROFILE_MISSING'
    WHEN pref."userId" IS NULL THEN 'PREFERENCE_MISSING'
    ELSE 'OK'
  END AS "gateStatus",
  true AS "includedInPool",
  NULL::text AS "blockerReason"
FROM onboarding_photo_preview_pools opp
JOIN onboarding_photo_preview_pool_items oppi ON oppi."poolId" = opp."id"
LEFT JOIN profile_dims pd ON pd."userId" = oppi."candidateUserId"
LEFT JOIN user_preferences pref ON pref."userId" = oppi."candidateUserId"
WHERE opp."status" = 'active' AND opp."userId" = ':viewer_id'
ORDER BY oppi."candidateUserId";
```

**验收：** `gateStatus` 全为 `OK` · **0** 行 `SELF_IN_POOL` · `photoVisualTagsCount > 0`（若 r3b 需要 tags）。

## 7. Route C 执行流程

```text
FOR each viewer IN RouteC_cohort (5–10):
  1. Run §4 readiness (single viewer filter)
  2. Run §5 candidate detail SQL
  3. IF vision missing → §8.1 backfill (then re-check)
  4. IF 20D/pref missing → swap viewer (no seed)
  5. IF pool < 6 → swap viewer (no synthetic pool)
  6. IF no MatchResult → §8.2 batch match (staging)
  7. r3b → artifacts/p76/r7g/<V>/r3b.json
  8. r4b (r3b candidate ids) → r4b.json
  9. r5b (r4b top2) → r5b.json
 10. r6b (stage JSON paths) → r6b.json
 11. Append §9 PM evidence row
END
Aggregate §10 → r7h closeout
```

| Blocker | 处理 |
|---------|------|
| r3b `GENDER_GATE` / `PREFERENCE_GATE` 系统性 | 换 viewer 或修真实 preference 人口字段（**不**关 gate） |
| r4b `CANDIDATE_PROFILE_MISSING` | 换 viewer（Route C 不应 seed） |
| r6b invalid JSON | 用 `cmd` 重定向或 UTF-8 no BOM（见 [r7c4](./P7.6-r7c4-fixture-preparation-route-a-closeout.md)） |
