| **GENDER_GATE** | 候选与 viewer 的**异性二进制门控**未通过（`User.gender` → `male`/`female`/`unknown`） | **不得绕过**；换 viewer 或确保 cohort 含足够异性用户 |

**为何 13 + 9 ≠ 20？** 单条 pair 可叠加多个 `ineligibleReasons`；adapter 扫描 20 人，分布为 reason **出现次数**之和。

### 4.0 公共 CTE：`usable_vision_images`

与 r3b adapter 对齐的「单图 usable vision」判定（简化版，供 A–E 复用）：

```sql
-- 复制到 psql 会话后，后续查询可 WITH usable_vision_images AS (...)
WITH usable_vision_images AS (
  SELECT
    ui.id AS image_id,
    ui."userId" AS user_id,
    ui."createdAt",
    ui."reviewStatus",
    ui."detectionStatus",
    (ui."detectionScoreJson"->'vision'->>'sourceVersion') AS vision_source_version,
    (ui."detectionScoreJson"->'vision'->>'visionStatus') AS vision_status,
    jsonb_array_length(
      COALESCE(ui."detectionScoreJson"->'vision'->'photoVisualTags', '[]'::jsonb)
    ) AS photo_visual_tag_count
  FROM user_images ui
  WHERE ui."detectionScoreJson" IS NOT NULL
    AND jsonb_typeof(ui."detectionScoreJson"->'vision') = 'object'
    AND (ui."detectionScoreJson"->'vision'->>'visionStatus') = 'ok'
    AND jsonb_array_length(
          COALESCE(ui."detectionScoreJson"->'vision'->'photoVisualTags', '[]'::jsonb)
        ) > 0
    AND COALESCE(ui."reviewStatus", '') NOT IN (
          'rejected', 'needs_reupload', 'appealed', 'appeal_rejected'
        )
    AND (
      ui."reviewStatus" IN ('approved', 'appeal_approved')
      OR (
        COALESCE(ui."reviewStatus", 'not_required') IN ('pending_review', 'not_required')
        AND ui."detectionStatus" IN ('passed', 'skipped')
      )
    )
)
SELECT * FROM usable_vision_images LIMIT 5;
```

### 4.A 找有 live vision + styleTags 的用户

```sql
WITH usable_vision_images AS (
  SELECT DISTINCT ON (ui."userId")
    ui."userId" AS user_id,
    (ui."detectionScoreJson"->'vision'->>'sourceVersion') AS vision_source_version
  FROM user_images ui
  WHERE (ui."detectionScoreJson"->'vision'->>'visionStatus') = 'ok'
    AND jsonb_array_length(COALESCE(ui."detectionScoreJson"->'vision'->'photoVisualTags', '[]'::jsonb)) > 0
    AND COALESCE(ui."reviewStatus", '') NOT IN ('rejected','needs_reupload','appealed','appeal_rejected')
    AND (
      ui."reviewStatus" IN ('approved','appeal_approved')
      OR (COALESCE(ui."reviewStatus",'not_required') IN ('pending_review','not_required')
          AND ui."detectionStatus" IN ('passed','skipped'))
    )
  ORDER BY ui."userId", ui."createdAt" DESC
)
SELECT
  u.id,
  u.gender,
  u."onboardingPhotoAestheticCompletedAt" IS NOT NULL AS aesthetic_done,
  cardinality(up."styleTags") AS style_tag_count,
  uvi.vision_source_version,
  EXISTS (SELECT 1 FROM user_profile p WHERE p."userId" = u.id) AS has_relation_profile
FROM users u
JOIN usable_vision_images uvi ON uvi.user_id = u.id
LEFT JOIN user_preferences up ON up."userId" = u.id
WHERE cardinality(COALESCE(up."styleTags", ARRAY[]::text[])) > 0
ORDER BY (uvi.vision_source_version = 'p7.5-r7-cloud-zhipu-v1') DESC, style_tag_count DESC
LIMIT 30;
```

### 4.C 按 gender 统计可能通过异性 gate 的候选数量

将 `:viewer_id` 换成实际 viewer（先查 gender）：

```sql
-- Step 1: viewer 性别
SELECT id, gender,
  CASE
    WHEN lower(trim(gender)) IN ('male','m','男') THEN 'male'
    WHEN lower(trim(gender)) IN ('female','f','女') THEN 'female'
    ELSE 'unknown'
  END AS gender_norm
FROM users
WHERE id = 'cmo7ksq8s00006znosryc9k0n';

-- Step 2: 异性 + relation_profile + 有图 的候选计数
WITH viewer AS (
  SELECT id,
    CASE
      WHEN lower(trim(gender)) IN ('male','m','男') THEN 'male'
      WHEN lower(trim(gender)) IN ('female','f','女') THEN 'female'
      ELSE 'unknown'
    END AS gender_norm
  FROM users
  WHERE id = 'cmo7ksq8s00006znosryc9k0n'
)
SELECT
  COUNT(*) FILTER (WHERE v.gender_norm = 'unknown') AS viewer_unknown_blocks_all,
  COUNT(*) FILTER (
    WHERE v.gender_norm = 'male'
      AND lower(trim(c.gender)) IN ('female','f','女')
  ) AS opposite_female_if_viewer_male,
  COUNT(*) FILTER (
    WHERE v.gender_norm = 'female'
      AND lower(trim(c.gender)) IN ('male','m','男')
  ) AS opposite_male_if_viewer_female
FROM viewer v
CROSS JOIN users c
WHERE c.id <> v.id
  AND EXISTS (SELECT 1 FROM user_images i WHERE i."userId" = c.id)
  AND EXISTS (SELECT 1 FROM user_profile p WHERE p."userId" = c.id);
```

### 4.D 对固定 viewer，gated 扫描范围内 VISION_NOT_OK 粗算

模拟 r3b 前 20 个 gated 候选（`createdAt` asc，与 adapter 一致），统计无 usable vision 的人数：

```sql
WITH viewer AS (
  SELECT id FROM users WHERE id = 'cmo7ksq8s00006znosryc9k0n'
),
gated AS (
  SELECT c.id AS candidate_id, c.gender
  FROM users c, viewer v
  WHERE c.id <> v.id
    AND EXISTS (SELECT 1 FROM user_images i WHERE i."userId" = c.id)
    AND EXISTS (SELECT 1 FROM user_profile p WHERE p."userId" = c.id)
  ORDER BY c."createdAt" ASC
  LIMIT 20
),
has_usable_vision AS (
  SELECT DISTINCT ui."userId"
  FROM user_images ui
  WHERE (ui."detectionScoreJson"->'vision'->>'visionStatus') = 'ok'
    AND jsonb_array_length(COALESCE(ui."detectionScoreJson"->'vision'->'photoVisualTags', '[]'::jsonb)) > 0
    AND COALESCE(ui."reviewStatus", '') NOT IN ('rejected','needs_reupload','appealed','appeal_rejected')
    AND (
      ui."reviewStatus" IN ('approved','appeal_approved')
      OR (COALESCE(ui."reviewStatus",'not_required') IN ('pending_review','not_required')
          AND ui."detectionStatus" IN ('passed','skipped'))
    )
)
SELECT
  COUNT(*) AS scanned,
  COUNT(*) FILTER (WHERE hv."userId" IS NULL) AS vision_not_ok_approx,
  COUNT(*) FILTER (WHERE hv."userId" IS NOT NULL) AS has_usable_vision
FROM gated g
LEFT JOIN has_usable_vision hv ON hv."userId" = g.candidate_id;
```

### 4.E 找「异性 + usable vision + styleTags」候选较多的 viewer

Heuristic：对每个 strict-binary viewer，数异性候选中同时满足 vision + styleTags 的人数（**不含** preference gate，需 r3b 最终确认）：

```sql
WITH usable_user AS (
  SELECT DISTINCT ui."userId" AS user_id
  FROM user_images ui
  WHERE (ui."detectionScoreJson"->'vision'->>'visionStatus') = 'ok'
    AND jsonb_array_length(COALESCE(ui."detectionScoreJson"->'vision'->'photoVisualTags', '[]'::jsonb)) > 0
    AND COALESCE(ui."reviewStatus", '') NOT IN ('rejected','needs_reupload','appealed','appeal_rejected')
    AND (
      ui."reviewStatus" IN ('approved','appeal_approved')
      OR (COALESCE(ui."reviewStatus",'not_required') IN ('pending_review','not_required')
          AND ui."detectionStatus" IN ('passed','skipped'))
    )
),
viewer_candidates AS (
  SELECT
    v.id AS viewer_id,
    v.gender AS viewer_gender,
    CASE
      WHEN lower(trim(v.gender)) IN ('male','m','男') THEN 'male'
      WHEN lower(trim(v.gender)) IN ('female','f','女') THEN 'female'
      ELSE 'unknown'
    END AS viewer_norm,
    c.id AS candidate_id
  FROM users v
  JOIN users c ON c.id <> v.id
  JOIN user_profile cp ON cp."userId" = c.id
  WHERE EXISTS (SELECT 1 FROM user_images i WHERE i."userId" = c.id)
    AND EXISTS (SELECT 1 FROM usable_user uu WHERE uu.user_id = v.id)
    AND EXISTS (SELECT 1 FROM user_preferences vp WHERE vp."userId" = v.id
                AND cardinality(vp."styleTags") > 0)
)
SELECT
  vc.viewer_id,
  vc.viewer_gender,
  COUNT(*) FILTER (
    WHERE vc.viewer_norm = 'male'
      AND lower(trim(c.gender)) IN ('female','f','女')
      AND EXISTS (SELECT 1 FROM usable_user uu WHERE uu.user_id = vc.candidate_id)
      AND EXISTS (SELECT 1 FROM user_preferences cp WHERE cp."userId" = vc.candidate_id
                  AND cardinality(cp."styleTags") > 0)
  ) AS opposite_with_vision_and_style_female_candidates,
  COUNT(*) FILTER (
    WHERE vc.viewer_norm = 'female'
      AND lower(trim(c.gender)) IN ('male','m','男')
      AND EXISTS (SELECT 1 FROM usable_user uu WHERE uu.user_id = vc.candidate_id)
      AND EXISTS (SELECT 1 FROM user_preferences cp WHERE cp."userId" = vc.candidate_id
                  AND cardinality(cp."styleTags") > 0)
  ) AS opposite_with_vision_and_style_male_candidates
FROM viewer_candidates vc
JOIN users c ON c.id = vc.candidate_id
WHERE vc.viewer_norm IN ('male','female')
GROUP BY vc.viewer_id, vc.viewer_gender, vc.viewer_norm
HAVING COUNT(*) FILTER (
    WHERE (vc.viewer_norm = 'male' AND lower(trim(c.gender)) IN ('female','f','女'))
       OR (vc.viewer_norm = 'female' AND lower(trim(c.gender)) IN ('male','m','男'))
  ) > 0
ORDER BY GREATEST(
  COUNT(*) FILTER (WHERE vc.viewer_norm = 'male' AND lower(trim(c.gender)) IN ('female','f','女')),
  COUNT(*) FILTER (WHERE vc.viewer_norm = 'female' AND lower(trim(c.gender)) IN ('male','m','男'))
) DESC
LIMIT 20;
```

**从 4.E 取 `viewer_id` top 1**，再跑 §8 r3b 命令验证。
